const TestimonialCarousel = () => {
  const marqueeRef = useRef(null);
  const [gsapLoaded, setGsapLoaded] = useState(false);

  useEffect(() => {
    const script = document.createElement('script');
    script.src = "https://cdnjs.cloudflare.com/ajax/libs/gsap/3.12.2/gsap.min.js";
    script.async = true;
    script.onload = () => setGsapLoaded(true);
    document.head.appendChild(script);
    
    return () => {
      document.head.removeChild(script);
    };
  }, []);

  useEffect(() => {
    if (!gsapLoaded || !window.gsap || !marqueeRef.current) return;

    const gsap = window.gsap;
    const marquee = marqueeRef.current;
    
    // Calcula a largura real de metade do conteúdo (já que duplicamos a lista)
    const scrollWidth = marquee.scrollWidth;
    
    const animation = gsap.to(marquee, {
      x: `-${scrollWidth / 2}`,
      duration: 35,
      ease: "none",
      repeat: -1,
    });

    const handleMouseEnter = () => gsap.to(animation, { timeScale: 0.15, duration: 0.6 });
    const handleMouseLeave = () => gsap.to(animation, { timeScale: 1, duration: 0.6 });

    marquee.addEventListener('mouseenter', handleMouseEnter);
    marquee.addEventListener('mouseleave', handleMouseLeave);

    return () => {
      animation.kill();
      marquee.removeEventListener('mouseenter', handleMouseEnter);
      marquee.removeEventListener('mouseleave', handleMouseLeave);
    };
  }, [gsapLoaded]);

  return (
    <section className="relative min-h-screen w-full flex flex-col justify-center overflow-hidden bg-[#070708] font-sans">
      
      {/* Background Orbs */}
      <div className="absolute top-[-10%] left-[-5%] w-[50%] h-[50%] bg-purple-600/15 rounded-full blur-[120px] animate-pulse"></div>
      <div className="absolute bottom-[-10%] right-[-5%] w-[50%] h-[50%] bg-blue-600/10 rounded-full blur-[120px] animate-pulse"></div>

      <div className="relative z-10 w-full">
        <div className="container mx-auto px-6 mb-12 text-center md:text-left">
          <div className="flex items-center gap-4 mb-4 justify-center md:justify-start">
            <div className="w-12 h-[1px] bg-purple-500"></div>
            <span className="text-purple-400 font-mono text-xs uppercase tracking-[0.4em]">Experiências Digitais</span>
          </div>
          <h2 className="text-5xl md:text-7xl font-bold text-white tracking-tighter leading-[0.9]">
            A voz de quem <br />
            <span className="text-transparent bg-clip-text bg-gradient-to-r from-purple-400 via-white to-blue-400">
              lidera o mercado.
            </span>
          </h2>
        </div>

        <div className="relative w-full overflow-hidden">
          {/* Fades laterais */}
          <div className="absolute inset-y-0 left-0 w-24 md:w-48 bg-gradient-to-r from-[#070708] to-transparent z-20 pointer-events-none"></div>
          <div className="absolute inset-y-0 right-0 w-24 md:w-48 bg-gradient-to-l from-[#070708] to-transparent z-20 pointer-events-none"></div>

          <div 
            ref={marqueeRef}
            className="flex whitespace-nowrap will-change-transform py-8"
          >
            {/* Duplicação para loop infinito */}
            {[...testimonials, ...testimonials].map((item, index) => (
              <TestimonialCard key={`${item.id}-${index}`} item={item} />
            ))}
          </div>
        </div>
      </div>

      {/* Marca d'água de fundo */}
      <div className="absolute bottom-10 right-10 opacity-[0.03] pointer-events-none select-none text-right">
        <p className="text-[12vw] font-black text-white leading-none">GLASS</p>
        <p className="text-[5vw] font-light text-white tracking-[1em] -mt-4 uppercase">Testimonials</p>

};
