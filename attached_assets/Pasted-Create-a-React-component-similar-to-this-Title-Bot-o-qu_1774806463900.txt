/**
 * MOTOR DE ÁUDIO SINTÉTICO (WEB AUDIO API)
 * Gera sons em tempo real sem arquivos externos
 */
class AudioSynth {
  constructor() {
    this.ctx = null;
    this.osc = null;
    this.gain = null;
    this.lfo = null;
    this.isInit = false;
  }

  init() {
    if (this.isInit) return;
    const AudioContext = window.AudioContext || window.webkitAudioContext;
    this.ctx = new AudioContext();
    this.isInit = true;
  }

  // Som de zumbido ambiente (Drone)
  playDrone(intensity) {
    if (!this.isInit) this.init();
    if (this.ctx.state === 'suspended') this.ctx.resume();

    if (!this.osc) {
      this.osc = this.ctx.createOscillator();
      this.gain = this.ctx.createGain();
      this.lfo = this.ctx.createOscillator();
      const lfoGain = this.ctx.createGain();

      this.osc.type = 'sawtooth';
      this.lfo.type = 'sine';
      this.lfo.frequency.value = 10; // Tremolo speed
      lfoGain.gain.value = 500; // Tremolo depth

      this.osc.connect(this.gain);
      this.lfo.connect(lfoGain);
      lfoGain.connect(this.osc.frequency);
      
      this.gain.connect(this.ctx.destination);
      this.osc.start();
      this.lfo.start();
      this.gain.gain.value = 0;
    }

    // Modula frequência e volume baseada na distância do mouse
    const now = this.ctx.currentTime;
    this.osc.frequency.setTargetAtTime(50 + (intensity * 100), now, 0.1);
    this.gain.gain.setTargetAtTime(intensity * 0.1, now, 0.1);
  }

  // Som de Carregamento (Riser)
  playCharge(progress) {
    if (!this.isInit || !this.osc) return;
    const now = this.ctx.currentTime;
    // Aumenta pitch drasticamente
    this.osc.frequency.setTargetAtTime(100 + (progress * 20), now, 0.05);
    // Adiciona distorção/volume
    this.gain.gain.setTargetAtTime(0.1 + (progress * 0.002), now, 0.05);
    // Acelera o LFO para criar tensão
    this.lfo.frequency.setTargetAtTime(10 + (progress * 0.5), now, 0.1);
  }

  // Som de Explosão
  triggerExplosion() {
    if (!this.isInit) return;
    const osc = this.ctx.createOscillator();
    const gain = this.ctx.createGain();
    
    osc.type = 'triangle';
    osc.frequency.setValueAtTime(200, this.ctx.currentTime);
    osc.frequency.exponentialRampToValueAtTime(0.01, this.ctx.currentTime + 1.5);
    
    gain.gain.setValueAtTime(0.5, this.ctx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.01, this.ctx.currentTime + 1.5);
    
    osc.connect(gain);
    gain.connect(this.ctx.destination);
    osc.start();
    osc.stop(this.ctx.currentTime + 1.5);
  }

  stop() {
    if (this.gain) {
      this.gain.gain.setTargetAtTime(0, this.ctx.currentTime, 0.5);
    }
  }
}

const QuantumReactor = () => {

  const [position, setPosition] = useState({ x: 0, y: 0 });

  const [isHovered, setIsHovered] = useState(false);

  const [charge, setCharge] = useState(0);

  // Efeito de Partículas (Loop)
  useEffect(() => {
    let animationFrame;
    const updateParticles = () => {
      setParticles(prev => 
        prev
          .map(p => {
            let newX = p.x;
            let newY = p.y;
            let newSpeed = p.speed;

            // Implosão (durante carga alta)
            if (p.type === 'implode') {
               const angleToCenter = Math.atan2(-p.y, -p.x);
               newX += Math.cos(angleToCenter) * p.speed;
               newY += Math.sin(angleToCenter) * p.speed;
               // Se chegar perto do centro, some
               if (Math.abs(newX) < 5 && Math.abs(newY) < 5) return null;
            } 
            // Explosão normal
            else {
               newX += Math.cos(p.angle) * p.speed;
               newY += Math.sin(p.angle) * p.speed;
               newSpeed *= 0.95; // atrito
            }

            return {
              ...p,
              x: newX,
              y: newY,
              speed: newSpeed,
              life: p.life - p.decay,
              size: p.size * (p.type === 'implode' ? 0.9 : 0.96)
            };
          })
          .filter(p => p !== null && p.life > 0)
      );
      animationFrame = requestAnimationFrame(updateParticles);
    };
    
    if (particles.length > 0) {
      animationFrame = requestAnimationFrame(updateParticles);
    }
    return () => cancelAnimationFrame(animationFrame);

  // Manipulação do Mouse (Tilt 3D + Magnetismo)

    synth.init();

    const centerX = rect.left + rect.width / 2;
    const centerY = rect.top + rect.height / 2;
    
    const x = e.clientX - centerX;
    const y = e.clientY - centerY;
    
    // Distância normalizada (0 a 1) para áudio
    const dist = Math.sqrt(x * x + y * y);
    const maxDist = 300;
    const intensity = Math.max(0, 1 - dist / maxDist);
    
    // Tocar Drone se não estiver carregando
    if (status !== 'charging') {
      synth.playDrone(intensity);
    }

    if (dist < maxDist) {
      // Efeito Tilt 3D
      const rotateY = (x / (rect.width / 2)) * 20; // Max 20deg
      const rotateX = -(y / (rect.height / 2)) * 20;

      setRotation({ x: rotateX, y: rotateY });
      setPosition({ x: x * 0.2, y: y * 0.2 }); // Movimento magnético
      setIsHovered(true);
    } else {
      setRotation({ x: 0, y: 0 });
      setPosition({ x: 0, y: 0 });
      setIsHovered(false);
      synth.stop();
    }
  };

  const handleMouseLeave = () => {
    setRotation({ x: 0, y: 0 });
    setPosition({ x: 0, y: 0 });
    setIsHovered(false);
    synth.stop();
    
    if (status === 'charging') {
      setStatus('idle');
      setCharge(0);
      setGlitchActive(false);
    }

          else setGlitchActive(false);

          if (newVal >= 100) {

            triggerSingularity();
            return 100;
          }

        });
